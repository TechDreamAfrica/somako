AliExpress Commands:

Jumia Commands:

# Mock mode (recommended for testing)
python manage.py scrape_jumia --mock --category "electronics" --count 20

# Real scraping by keyword
python manage.py scrape_jumia --keyword "smartphones" --count 20

# Real scraping by category  
python manage.py scrape_jumia --category "electronics" --count 50

# Real scraping all categories
python manage.py scrape_jumia --all-categories --count 10

# With custom settings
python manage.py scrape_jumia --keyword "laptops" --count 25 --db-category electronics --user admin
